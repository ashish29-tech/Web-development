function a(){
  let count = 10;
  function b(){
    console.log(count);
  }
  return b; //→ a is a Higher-Order Function.
}

let x = a(); //Now x contains the function b. Matlab x() = b().
x(); // closure example. Even though a() finish ho chuka, b() still remembers count = 10
